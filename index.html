document.getElementById('downloadQR').addEventListener('click', async () => { const box = document.getElementById('qrImage'); const message = document.getElementById('qrDownloadMessage'); const canvas = box.querySelector('canvas'); const img = box.querySelector('img'); const id = document.getElementById('qrTokenId').textContent.slice(-3); try { let dataUrl; if (canvas && canvas.width > 0) { dataUrl = canvas.toDataURL('image/png'); } else if (img) { if (!img.complete) { await new Promise((resolve, reject) => { img.onload = resolve; img.onerror = reject; }); } const c = document.createElement('canvas'); c.width = img.naturalWidth; c.height = img.naturalHeight; c.getContext('2d').drawImage(img, 0, 0); dataUrl = c.toDataURL('image/png'); } else { throw new Error('QR not ready'); } const a = document.createElement('a'); a.href = dataUrl; a.download = `QZIX-COIN-Token-${id}.png`; document.body.appendChild(a); a.click(); a.remove(); message.textContent = 'QR download requested. Check your Downloads folder.'; message.style.color = 'var(--accent)'; } catch (error) { message.textContent = 'Download failed. Open QR and save image manually.'; message.style.color = 'var(--danger)'; if (img) window.open(img.src, '_blank'); } });
